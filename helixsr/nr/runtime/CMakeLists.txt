cmake_minimum_required(VERSION 3.24)
project(HelixSrNr VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This backend targets D3D12. Use MSVC or a Windows cross compiler.")
endif()
option(HELIXSR_NR_BUILD_PROBE "Build the GPU/lifetime validation caller and diagnostic endpoint export" OFF)
find_program(DXC_EXECUTABLE NAMES dxc REQUIRED)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(frame_headers)
foreach(entry FramePre FramePost)
    set(header "${CMAKE_CURRENT_BINARY_DIR}/generated/${entry}.h")
    add_custom_command(OUTPUT "${header}"
        COMMAND "${DXC_EXECUTABLE}" -T cs_6_2 -E "${entry}" -Gis -HV 2018
            -I "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/Frame.hlsl"
            -Fo "${CMAKE_CURRENT_BINARY_DIR}/generated/${entry}.cso"
            -Fh "${header}" -Vn "HelixSrNr${entry}"
        DEPENDS Frame.hlsl FrameConstants.h VERBATIM)
    list(APPEND frame_headers "${header}")
endforeach()
add_library(helixsr-nr SHARED Graph.cpp Runtime.cpp ${frame_headers})
target_compile_features(helixsr-nr PRIVATE cxx_std_17)
target_precompile_headers(helixsr-nr PRIVATE pch.h)
target_compile_definitions(helixsr-nr PRIVATE HELIXSR_NR_BUILD NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(helixsr-nr PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(helixsr-nr PRIVATE d3d12 dxgi dxguid bcrypt)
set_target_properties(helixsr-nr PROPERTIES PREFIX "" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
    target_compile_options(helixsr-nr PRIVATE /W4 /permissive- /EHsc)
elseif(MINGW)
    target_link_options(helixsr-nr PRIVATE -static -pthread)
endif()
